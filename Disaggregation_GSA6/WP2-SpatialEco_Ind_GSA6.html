Fuel costs

The following code block produces a bar plot showing fuel costs by country and year.

AER_wide <- AER %>%
  dplyr::group_by(country_name, supra_region, gear_type, year, vessel_length, variable_name) %>%
  dplyr::summarise(total_value = sum(value, na.rm = TRUE)) %>%
  pivot_wider(names_from  = variable_name,
              values_from = total_value)

Fuel_cost <- AER_wide %>%
  group_by(country_name,supra_region, gear_type, year) %>% 
  summarise(Energy_cost_sum = sum(Energy_costs),
            Energy_consumption_sum = sum(Energy_consumption)) %>% 
  mutate(Fuel_cost = Energy_cost_sum / Energy_consumption_sum)


ggplot()+

Redistribute economic variables

Computes the total fishing days for each economic grouping. Distributes each economic variable across c-squares proportionally to fishing effort (fishing days) in each cell.

FDI_effort_CS <- FDI_effort_sf %>%
                st_drop_geometry() %>% 
                group_by(cscode,year,vessel_length, gear_type,supra_region) %>%
                summarise(totfishdays = round(sum(totfishdays, na.rm = TRUE),1)) %>% 
                group_by(year,vessel_length, gear_type,supra_region) %>%
                mutate(sum_days = sum(totfishdays)) %>% 
                mutate(effort_share = ifelse(sum_days > 0,
                              totfishdays / sum_days,
                              0))

aer_fdi <- FDI_effort_CS %>%
            inner_join(AER_wide, by = c("year" ,"vessel_length", "gear_type", "supra_region") )

aer_fdi_disagg <- aer_fdi %>%
   mutate( GVL_disagg  = Gross_value_of_landings * effort_share,
    Other_income_disagg = Other_income * effort_share,
    Repair_maintenance_costs_disagg = Repair_maintenance_costs*effort_share,
    Other_variable_costs_disagg = Other_variable_costs * effort_share,
    Other_non_variable_costs_disagg = Other_non_variable_costs * effort_share,
    Personnel_costs_disagg =  Personnel_costs * effort_share,
    Value_of_unpaid_labour_disagg = Value_of_unpaid_labour * effort_share) %>% 
   
   select(cscode, year ,vessel_length, gear_type, supra_region, GVL_disagg, Other_income_disagg, Repair_maintenance_costs_disagg,
         Other_variable_costs_disagg, Other_non_variable_costs_disagg, Personnel_costs_disagg, Value_of_unpaid_labour_disagg,
         totfishdays, Fuel_cost)

Applying Sala et al. fuel consumption model

Defines the parameters of the Sala model: Table 8 Sala et al., 2022 dFC = m × LOA + q → daily fuel consumption as a function of vessel length. (https://www.nature.com/articles/s41597-022-01478-0)

Fuel_coefficients = data.frame(gear_type = c("OTB"),
                               m = c(1.470),
                               q = c(12.811))

Adds the fuel model parameters and assigns an average LOA to length classes. If you have VMS data and MMSI you can have more detailed LOA info

x = aer_fdi_disagg
x = left_join(x, Fuel_coefficients, relationship = "many-to-many")

x = x %>% mutate(loa = recode(vessel_length, 
                              "VL1824" = 21,
                              "VL1218" = 15,
                              "VL2440" = 32,
                              "VL0612" = 9))

Computes:

  • daily fuel consumption = dFC = m ⋅ LOA + q

  • annual consumption = FC = dFC * Total Fishing days (L/year)

  • annual energy cost per cell = FC * fuel cost

x$loaFC = x$q + (x$loa*x$m) 
x$yFC = x$loaFC * x$totfishdays
x$EC = x$yFC * x$Fuel_cost

Compute spatial GVA and Gross Profits

Re-computes GVA and Gross Profit after including the Sala-based energy costs.

  • GVA = GVL + Other_income - EC - Repair_maintenance_costs - Other_variable_costs - Other_non_variable_costs;

  • Gross_profit = GVL + Other_income - Personnel_costs - Value_of_unpaid_labour - EC - Repair_maintenance_costs - Other_variable_costs - Other_non_variable_costs

x =   x %>% 
  mutate(
    GVA = GVL_disagg + Other_income_disagg -
      EC - Repair_maintenance_costs_disagg -
      Other_variable_costs_disagg - Other_non_variable_costs_disagg,
    
    Gross_profit = GVL_disagg + Other_income_disagg -
      Personnel_costs_disagg - Value_of_unpaid_labour_disagg -
      EC - Repair_maintenance_costs_disagg -
      Other_variable_costs_disagg - Other_non_variable_costs_disagg)
